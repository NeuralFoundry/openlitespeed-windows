cmake_minimum_required(VERSION 3.16)
project(yajl C)
set(YAJL_MAJOR 2)
set(YAJL_MINOR 1)
set(YAJL_MICRO 0)
set(INC ${CMAKE_CURRENT_BINARY_DIR}/include/yajl)
configure_file(${YAJL_SRC}/src/api/yajl_version.h.cmake ${INC}/yajl_version.h)
file(GLOB HDRS ${YAJL_SRC}/src/api/*.h)
file(COPY ${HDRS} DESTINATION ${INC})
file(GLOB SRCS ${YAJL_SRC}/src/*.c)
add_library(yajl STATIC ${SRCS})
target_include_directories(yajl PRIVATE ${YAJL_SRC}/src ${CMAKE_CURRENT_BINARY_DIR}/include)
target_compile_definitions(yajl PRIVATE _CRT_SECURE_NO_WARNINGS)
install(TARGETS yajl ARCHIVE DESTINATION lib)
install(FILES ${HDRS} ${INC}/yajl_version.h DESTINATION include/yajl)
